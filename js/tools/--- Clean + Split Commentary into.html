// --- Clean + Split Commentary into JSON Paragraphs --- function createWeekJsonFromPaste(rawText, weekNumber) { if (!rawText || typeof rawText !== "string") { console.error("No text provided!"); return null; } // Step 1: Clean smart quotes and weird WordPress symbols let cleanText = rawText .replace(/[\u2018\u2019\u201A\u201B\u2032\u2035]/g, "'") // single quotes .replace(/[\u201C\u201D\u201E\u201F\u2033\u2036]/g, '"') // double quotes .replace(/\u00A0/g, " ") // non-breaking spaces .replace(/ /g, " ") .replace(/<\/?[^>]+(>|$)/g, "") // strip HTML tags .trim(); // Step 2: Split into paragraphs (by double newlines or line breaks) const paragraphs = cleanText .split(/\n\s*\n+/) .map(p => p.trim()) .filter(p => p.length > 0); // Step 3: Structure as JSON const weekData = { week: weekNumber || 1, paragraphs: paragraphs.map((text, i) => ({ id: i + 1, text })) }; console.log("✅ Week JSON created:", weekData); return weekData; } // Example usage: const sample = ` This is the first paragraph copied from WordPress. Here is the second one, with extra spacing. And a third paragraph. `; const weekJson = createWeekJsonFromPaste(sample, 3); console.log(JSON.stringify(weekJson, null, 2));
