// Global Audio Player Singleton
window.globalAudio = new Audio();
window.globalAudio.preload = "auto";
window.globalAudio.volume = 1.0;

const floatingPlayer = document.getElementById("floatingPlayer");
const playButton = document.getElementById("floatingPlayButton");
const titleEl = document.getElementById("floatingPlayerTitle");
const progressBar = document.getElementById("floatingProgress");

// --- Utility: Show/Hide floating player ---
function showFloatingPlayer() {
  floatingPlayer.classList.remove("translate-y-32", "opacity-0", "pointer-events-none");
  floatingPlayer.classList.add("translate-y-0", "opacity-100", "pointer-events-auto");
}

function hideFloatingPlayer() {
  floatingPlayer.classList.add("translate-y-32", "opacity-0", "pointer-events-none");
  floatingPlayer.classList.remove("translate-y-0", "opacity-100", "pointer-events-auto");
}

// --- Core Play Logic ---
function playAudio(src, title = "Untitled") {
  if (!src) return;

  // Update player UI
  titleEl.textContent = title;
  progressBar.style.width = "0%";

  // Load & play
  window.globalAudio.src = src;
  window.globalAudio.play().catch(err => console.error("Playback error:", err));

  // Show the player
  showFloatingPlayer();
}

// --- Toggle Play/Pause ---
playButton.addEventListener("click", () => {
  if (window.globalAudio.paused) {
    window.globalAudio.play();
    playButton.textContent = "⏸";
  } else {
    window.globalAudio.pause();
    playButton.textContent = "▶";
  }
});

// --- Sync button state on play/pause ---
window.globalAudio.addEventListener("play", () => {
  playButton.textContent = "⏸";
});

window.globalAudio.addEventListener("pause", () => {
  playButton.textContent = "▶";
});

// --- Progress bar update ---
window.globalAudio.addEventListener("timeupdate", () => {
  if (window.globalAudio.duration) {
    const progress = (window.globalAudio.currentTime / window.globalAudio.duration) * 100;
    progressBar.style.width = ${progress}%;
  }
});

// --- Listen for chapter outline play buttons (rendered by mainLoader.js) ---
document.addEventListener("click", (e) => {
  const btn = e.target.closest("[data-audio-src]");
  if (btn) {
    const src = btn.getAttribute("data-audio-src");
    const title = btn.getAttribute("data-title") || "Untitled";
    playAudio(src, title);
  }
});

// Optional: hide when audio ends
window.globalAudio.addEventListener("ended", () => {
  progressBar.style.width = "100%";
  setTimeout(() => {
    hideFloatingPlayer();
  }, 1500);
});
