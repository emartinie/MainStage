<script>
(async function() {
  // --- Calendar Logic (same as before) ---
  const calendars = {
    hillel: [{ name: "Passover", date: "2025-04-13" }, { name: "Pentecost", date: "2025-06-01" }],
    enochian: [{ name: "Enoch Feast", date: "2025-05-15" }],
    zedokian: [{ name: "Zedokian Feast", date: "2025-07-20" }],
    orthodox: [{ name: "Orthodox Easter", date: "2025-04-20" }],
    msChristian: [{ name: "MS Christmas", date: "2025-12-25" }]
  };

  let currentCalendar = "hillel";

  function formatDate(d) { return d.toISOString().split("T")[0]; }
  function getNextFeast(calendar) {
    const now = new Date();
    const calFeasts = calendars[calendar];
    return calFeasts.find(f => new Date(f.date) > now) || calFeasts[0];
  }

  function renderFeastCountdown() {
    const feast = getNextFeast(currentCalendar);
    const feastDate = new Date(feast.date + "T18:00:00");
    function updateCountdown() {
      const now = new Date();
      const diff = feastDate - now;
      let text;
      if (diff <= 0) {
        text = ${feast.name} is today or over!;
      } else {
        const days = Math.floor(diff / 86400000);
        const hours = Math.floor((diff % 86400000) / 3600000);
        const mins = Math.floor((diff % 3600000) / 60000);
        const secs = Math.floor((diff % 60000) / 1000);
        text = ${days}d ${hours}h ${mins}m ${secs}s until ${feast.name};
      }
      document.getElementById("feastDisplay").innerText = text;
      document.getElementById("nextHolyDayText").innerText = feast.name;
    }
    updateCountdown();
    setInterval(updateCountdown, 1000);
  }

  function renderWeek() {
    const start = new Date();
    start.setDate(start.getDate() - start.getDay());
    const week = Array.from({length:7}, (_,i) => { const d=new Date(start); d.setDate(start.getDate()+i); return d; });
    document.getElementById("weekDisplay").innerHTML = "<strong>Week:</strong> "+week.map(d=>formatDate(d)).join(" • ");
  }

  function renderSabbath() {
    const week = Array.from({length:7}, (_,i)=>{ const d=new Date(); d.setDate(d.getDate()-d.getDay()+i); return d; });
    const sabbath = week.find(d=>d.getDay()===6);
    const today = new Date();
    document.getElementById("sabbathDisplay").innerHTML = <strong>Sabbath:</strong> ${formatDate(sabbath)} ${today.getDay()===6?"(Today!)":""};
  }

  document.querySelectorAll(".calendarFilterBtn").forEach(btn=>{
    btn.addEventListener("click", ()=> {
      currentCalendar = btn.dataset.calendar;
      renderFeastCountdown();
    });
  });

  renderWeek();
  renderSabbath();
  renderFeastCountdown();

  // --- Google Map Logic ---
  // Load your exported map points JSON file
  const mapPoints = await fetch("mapPoints.json").then(res=>res.json());

  function initMap() {
    const center = { lat: 31.7683, lng: 35.2137 }; // Example: Jerusalem
    const map = new google.maps.Map(document.getElementById("map"), {
      zoom: 5,
      center
    });

    mapPoints.forEach(p=>{
      const marker = new google.maps.Marker({
        position: {lat:p.lat, lng:p.lng},
        map,
        title: p.name
      });

      const infoWindow = new google.maps.InfoWindow({
        content: <strong>${p.name}</strong><br/>${p.info||""}
      });

      marker.addListener("click", ()=>infoWindow.open(map, marker));
    });
  }

  window.initMap = initMap; // required by Google Maps API
})();
</script>

<script>
(function() {
  const feastData = {
    biblicalFeasts: [
      { name: "Passover", date: "2025-04-13", info: "Passover details..." },
      { name: "Pentecost", date: "2025-06-01", info: "Pentecost details..." },
      { name: "Trumpets", date: "2025-09-23", info: "Trumpets details..." },
      { name: "Day of Atonement", date: "2025-10-02", info: "Day of Atonement details..." },
      { name: "Feast of Tabernacles", date: "2025-10-07", info: "Feast of Tabernacles details..." },
    ],
    otherReligions: [
      { name: "Gregorian Calendar Event", date: "2025-09-21", info: "Example event info..." },
      { name: "Islamic Calendar Event", date: "2025-09-23", info: "Example event info..." }
    ]
  };

  // Helpers
  function formatDate(d) {
    return d.toISOString().split("T")[0];
  }
  function getWeek(date = new Date()) {
    const start = new Date(date);
    start.setDate(date.getDate() - date.getDay());
    return Array.from({length: 7}, (_, i) => {
      const d = new Date(start);
      d.setDate(start.getDate() + i);
      return d;
    });
  }

  // Render Week
  function renderWeek() {
    const week = getWeek();
    document.getElementById("weekDisplay").innerHTML =
      "<strong>Week:</strong> " + week.map(d => formatDate(d)).join(" • ");
  }

  // Render Sabbath
  function renderSabbath() {
    const sabbath = getWeek().find(d => d.getDay() === 6);
    const today = new Date();
    document.getElementById("sabbathDisplay").innerHTML =
      <strong>Sabbath:</strong> ${formatDate(sabbath)} ${today.getDay() === 6 ? "(Today!)" : ""};
  }

  // Render Feasts / Countdown
  let countdownInterval = null;
  function renderFeast(view = "biblicalFeasts") {
    const nowStr = formatDate(new Date());
    const feastList = feastData[view];
    const nextFeast = feastList.find(f => f.date >= nowStr) || feastList[0];

    document.getElementById("feastDisplay").innerHTML =
      <strong>Next Feast:</strong> ${nextFeast.name} (${nextFeast.date});

    // Render Details
    const detailsDiv = document.getElementById("feastDetails");
    detailsDiv.innerHTML = feastList.map(f => <div class="p-1 border-b">${f.name}: ${f.info}</div>).join("");

    // Clear previous countdown
    if(countdownInterval) clearInterval(countdownInterval);

    // Start Countdown
    function updateCountdown() {
      const target = new Date(nextFeast.date + "T00:00:00");
      const diff = target - new Date();
      if(diff < 0) {
        document.getElementById("countdownDisplay").innerText = "Today!";
        return;
      }
      const days = Math.floor(diff/86400000);
      const hours = Math.floor((diff%86400000)/3600000);
      const mins = Math.floor((diff%3600000)/60000);
      const secs = Math.floor((diff%60000)/1000);
      document.getElementById("countdownDisplay").innerText =
        Countdown: ${days}d ${hours}h ${mins}m ${secs}s;
    }
    updateCountdown();
    countdownInterval = setInterval(updateCountdown, 1000);
  }

  // Filter Buttons
  document.querySelectorAll(".filterBtn").forEach(btn => {
    btn.addEventListener("click", () => {
      renderFeast(btn.dataset.view);
    });
  });

  // Initial render
  renderWeek();
  renderSabbath();
  renderFeast("biblicalFeasts");
})();
</script>

        <div class="feastHeader">${f.name} (${f.date})</div>
        <div class="feastInfo">${f.info}</div>
      </div>
    `).join("");

    // Collapsible feast details
    detailsDiv.querySelectorAll(".feastHeader").forEach(h => {
      h.style.cursor = "pointer";
      h.onclick = () => {
        const info = h.nextElementSibling;
        info.style.maxHeight = info.style.maxHeight ? null : info.scrollHeight + "px";
      };
    });

    if(countdownInterval) clearInterval(countdownInterval);

    function updateCountdown() {
      const target = new Date(nextFeast.date + "T00:00:00");
      const diff = target - new Date();
      if(diff < 0) {
        document.getElementById("countdownDisplay").innerText = "Today!";
        return;
      }
      const days = Math.floor(diff/86400000);
      const hours = Math.floor((diff%86400000)/3600000);
      const mins = Math.floor((diff%3600000)/60000);
      const secs = Math.floor((diff%60000)/1000);

      // Flip animation effect
      document.getElementById("countdownDisplay").innerHTML = `
        Countdown: 
        <span class="flip">${days}d</span> 
        <span class="flip">${hours}h</span> 
        <span class="flip">${mins}m</span> 
        <span class="flip">${secs}s</span>
      `;
    }

    updateCountdown();
    countdownInterval = setInterval(updateCountdown, 1000);
  }

  document.querySelectorAll(".filterBtn").forEach(btn => {
    btn.addEventListener("click", () => renderFeast(btn.dataset.view));
  });

  renderWeek();
  renderSabbath();
  renderFeast("biblicalFeasts");
})();
</script>

<script>
(function() {
  // Example static feasts with approximate sunset start/end offsets (in hours)
  const feasts = [
    { name: "Passover", date: "2025-04-13", sunsetOffset: -6, info: "Passover details..." },
    { name: "Pentecost", date: "2025-06-01", sunsetOffset: -6, info: "Pentecost details..." },
    { name: "Trumpets", date: "2025-09-23", sunsetOffset: -6, info: "Trumpets details..." },
    { name: "Day of Atonement", date: "2025-10-02", sunsetOffset: -6, info: "Day of Atonement details..." },
    { name: "Feast of Tabernacles", date: "2025-10-07", sunsetOffset: -6, info: "Feast of Tabernacles details..." }
  ];

  function formatDate(d) { return d.toISOString().split("T")[0]; }
  
  // Calculate sunset-based start/end for each feast
  function getSunsetDate(feast) {
    const d = new Date(feast.date + "T18:00:00"); // 18:00 = rough sunset
    d.setHours(d.getHours() + feast.sunsetOffset); // adjust if needed
    return d;
  }

  function renderFeastCountdown() {
    const now = new Date();
    const upcomingFeast = feasts.find(f => getSunsetDate(f) > now) || feasts[0];
    const feastStart = getSunsetDate(upcomingFeast);
    const feastEnd = new Date(feastStart);
    feastEnd.setDate(feastEnd.getDate() + 1); // ends next sunset

    function updateCountdown() {
      const diff = feastStart - new Date();
      let countdownText;
      if(diff <= 0 && new Date() < feastEnd) {
        countdownText = "Feast Today!";
      } else if(diff <= 0) {
        countdownText = "Feast Over";
      } else {
        const days = Math.floor(diff / 86400000);
        const hours = Math.floor((diff % 86400000) / 3600000);
        const mins = Math.floor((diff % 3600000) / 60000);
        const secs = Math.floor((diff % 60000) / 1000);
        countdownText = ${days}d ${hours}h ${mins}m ${secs}s until ${upcomingFeast.name};
      }
      document.getElementById("feastDisplay").innerText = countdownText;
    }

    updateCountdown();
    setInterval(updateCountdown, 1000);
  }

  function renderWeek() {
    const start = new Date();
    start.setDate(start.getDate() - start.getDay());
    const week = Array.from({length: 7}, (_, i) => {
      const d = new Date(start);
      d.setDate(start.getDate() + i);
      return d;
    });
    document.getElementById("weekDisplay").innerHTML =
      "<strong>Week:</strong> " + week.map(d => formatDate(d)).join(" • ");
  }

  function renderSabbath() {
    const week = Array.from({length: 7}, (_, i) => {
      const d = new Date();
      d.setDate(d.getDate() - d.getDay() + i);
      return d;
    });
    const sabbath = week.find(d => d.getDay() === 6);
    const today = new Date();
    document.getElementById("sabbathDisplay").innerHTML =
      <strong>Sabbath:</strong> ${formatDate(sabbath)} ${today.getDay() === 6 ? "(Today!)" : ""};
  }

  renderWeek();
  renderSabbath();
  renderFeastCountdown();
})();
</script>
