<!-- Zipcode filter input -->
<div class="filter-controls p-2 bg-gray-800 text-white rounded mb-2">
  <label for="zipInput">Filter by Zipcode:</label>
  <input id="zipInput" type="text" placeholder="Enter Zipcode" 
         class="px-2 py-1 text-black rounded"/>
  <button id="zipBtn" class="ml-2 px-2 py-1 bg-blue-600 rounded">Filter</button>
  <button id="resetBtn" class="ml-2 px-2 py-1 bg-gray-600 rounded">Reset</button>
</div>

<!-- Your map container -->
<div id="map" style="height: 400px;"></div>

<script>
// Example marker data (replace with your real dataset)
const markerData = [
  { lat: 37.7749, lng: -122.4194, title: "San Francisco", zipcode: "94103" },
  { lat: 34.0522, lng: -118.2437, title: "Los Angeles", zipcode: "90012" },
  { lat: 40.7128, lng: -74.0060, title: "New York", zipcode: "10001" }
];

// Setup Leaflet map
const map = L.map("map").setView([37.7749, -122.4194], 4);
L.tileLayer("https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png", {
  attribution: '&copy; OpenStreetMap contributors'
}).addTo(map);

// Layer group to hold markers
let markersLayer = L.layerGroup().addTo(map);

// Function to render markers
function renderMarkers(data) {
  markersLayer.clearLayers();
  data.forEach(item => {
    L.marker([item.lat, item.lng])
      .bindPopup(<b>${item.title}</b><br/>Zip: ${item.zipcode})
      .addTo(markersLayer);
  });
}

// Initial load
renderMarkers(markerData);

// Filter logic
document.getElementById("zipBtn").addEventListener("click", () => {
  const zip = document.getElementById("zipInput").value.trim();
  if (!zip) return;
  const filtered = markerData.filter(item => item.zipcode === zip);
  renderMarkers(filtered);
  if (filtered.length > 0) {
    map.fitBounds(filtered.map(f => [f.lat, f.lng]));
  }
});

document.getElementById("resetBtn").addEventListener("click", () => {
  renderMarkers(markerData);
  map.setView([37.7749, -122.4194], 4);
});
</script>
