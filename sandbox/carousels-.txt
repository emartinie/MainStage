    const cards = document.querySelectorAll(".bibleCard");

    function showCard(index) {
      cards.forEach((card, i) => {
        card.classList.add("inactive");
        card.classList.remove("active");
        if (i === index) {
          card.classList.add("active");
          card.classList.remove("inactive");
        }
      });
    }

    document.getElementById("biblePrevBtn").addEventListener("click", () => {
      current = (current - 1 + cards.length) % cards.length;
      showCard(current);
    });

      current = (current + 1) % cards.length;
      showCard(current);
    });

<script>
  const track = document.getElementById("bibleTrack");
  const cards = document.querySelectorAll(".bibleCard");
  const prevBtn = document.getElementById("biblePrev");
  const nextBtn = document.getElementById("bibleNext");

  let index = 0;
  let autoplayInterval = null;
  const autoplayDelay = 5000; // 5 seconds

  function updateStage() {
    track.style.transform = translateX(-${index * 100}%);
  }

  function nextCard() {
    index = (index + 1) % cards.length;
    updateStage();
  }

  function prevCard() {
    index = (index - 1 + cards.length) % cards.length;
    updateStage();
  }

  nextBtn.addEventListener("click", () => {
    nextCard();
    resetAutoplay();
  });

  prevBtn.addEventListener("click", () => {
    prevCard();
    resetAutoplay();
  });

  // Auto-play
  function startAutoplay() {
    autoplayInterval = setInterval(nextCard, autoplayDelay);
  }

  function stopAutoplay() {
    clearInterval(autoplayInterval);
    autoplayInterval = null;
  }

  function resetAutoplay() {
    stopAutoplay();
    startAutoplay();
  }

  // Pause on hover
  const stage = document.getElementById("bibleStage");
  stage.addEventListener("mouseenter", stopAutoplay);
  stage.addEventListener("mouseleave", startAutoplay);

  // Start auto-play initially
  startAutoplay();
</script>
