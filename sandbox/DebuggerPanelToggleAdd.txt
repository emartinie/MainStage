<!-- Debugger Panel Toggle -->
<button id="toggleDebugger" class="debug-btn">Toggle Debugger</button>
<div id="debuggerPanel" style="display:none;"></div>
</div> 

<script>
// Debugging helpers
const debug = (msg) => {
  const panel = document.getElementById('debuggerPanel');
  if(panel){ panel.innerHTML += msg + '<br>'; panel.scrollTop = panel.scrollHeight; }
};

// Toggle debugger
document.getElementById('toggleDebugger').addEventListener('click', () => {
  const panel = document.getElementById('debuggerPanel');
  panel.style.display = panel.style.display === 'none' ? 'block' : 'none';
});

// Example floating player playlist
document.addEventListener("DOMContentLoaded", () => {
  debug("DOM fully loaded, initializing sandbox...");

  // Dummy playlist
  const playlist = [
    { title: "Psalm 23", eng: "audio/Matthew01-Greek.mp3", heb: "audio/Matthew01-Greek.mp3" },
    { title: "John 3:16", eng: "audio/Matthew01-Greek.mp3", grk: "Matthew01-Greek.mp3" }
  ];

  // Dispatch playlist event
  window.dispatchEvent(new CustomEvent("player:updatePlaylist", { detail: { playlist } }));
  debug("Playlist dispatched to floating player.");

  // Log basic environment info
  debug`(Screen: ${window.innerWidth}x${window.innerHeight})`;
  debug`(User Agent: ${navigator.userAgent})`;
});
</script>
