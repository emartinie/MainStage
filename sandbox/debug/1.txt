      <header>
      <strong>Sandbox Debug</strong><span style="margin-left:auto" class="small" id="dbg-time">—</span></header>        
    <button id="closeDebuggerBtn">✖</button>    
    <div id="debug-summary" class="small"></div>
    <div id="debugOverlayLogs" role="log" aria-live="polite"></div>
    <div class="dbg-controls">
      <button id="dbg-clear">Clear</button>
      <button id="dbg-check-res">Check resources</button>
      <button id="dbg-export">Export logs</button>
      <button id="dbg-toggle">Toggle debug</button>
      <div id="customDebuggerTabs">
        <button data-tab="logs" class="active">Logs</button>
        <button data-tab="network">Network</button>
        <button data-tab="actions">Actions</button>
      </div>
      <div id="customDebuggerContent">
        <div id="customDebuggerLogs"></div>

<script>
/* ============================
   lightweight debug logger
   ============================ */
(function(){
  window.__dbg = { logs: [] };
  const out = document.getElementById('debugOverlayLogs');
  const summary = document.getElementById('debug-summary');
  function refreshSummary(){
    summary.textContent = navigator.userAgent;
    document.getElementById('dbg-time').textContent = new Date().toLocaleTimeString();
  }
  refreshSummary();
  function addLog(level,msg){
    const row = document.createElement('div');
    row.className = 'dbg-row ' + (level === 'error' ? 'dbg-level-error' : level === 'warn' ? 'dbg-level-warn' : 'dbg-level-info');
    row.textContent = '['+level.toUpperCase()+'] ' + msg;
    out.insertBefore(row, out.firstChild);
    window.__dbg.logs.push({t:Date.now(),level,msg});
    // cap
    if(window.__dbg.logs.length>200) window.__dbg.logs.shift();
  }
  // override console
  const nativeLog = console.log, nativeWarn = console.warn, nativeErr = console.error;
  console.log = function(...args){ nativeLog.apply(console,args); addLog('info', args.map(a=>String(a)).join(' ')); };
  console.warn = function(...args){ nativeWarn.apply(console,args); addLog('warn', args.map(a=>String(a)).join(' ')); };
  console.error = function(...args){ nativeErr.apply(console,args); addLog('error', args.map(a=>String(a)).join(' ')); };
  // global errors
  window.addEventListener('error', function(ev){ addLog('error', ev.message + ' — ' + (ev.filename||'') + ':' + (ev.lineno||'') ); });
  window.addEventListener('unhandledrejection', function(ev){ addLog('error', 'UnhandledPromiseRejection: '+ (ev.reason && ev.reason.message ? ev.reason.message : String(ev.reason))); });

  document.getElementById('dbg-clear').addEventListener('click', function(){ out.innerHTML=''; window.__dbg.logs=[]; });
  document.getElementById('dbg-check-res').addEventListener('click', function(){ checkResources(); });
  document.getElementById('dbg-export').addEventListener('click', function(){ 
    const blob = new Blob([JSON.stringify(window.__dbg.logs, null, 2)], {type:'application/json'}); 
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a'); a.href=url; a.download='dbg-logs.json'; a.click(); URL.revokeObjectURL(url);
  });
  document.getElementById('dbg-toggle').addEventListener('click', function(){ document.getElementById('debug-overlay').classList.toggle('hidden'); });

  // resource checker (simple)
  function checkResources(){
    addLog('info','Checking key resources...');
    const checks = [
      {type:'css', sel:'link[href*="css/cosmicbadass.css"]'},
      {type:'js', sel:'script[src*="js/FloatingPlayerv-3.js"]'},
      {type:'js', sel:'script[src*="js/mainStageLoader"]'}
    ];
    checks.forEach(c=>{
      const found = document.querySelector(c.sel);
      addLog('info', (found ? 'FOUND' : 'MISSING') + ' ' + c.type + ' selector: '+c.sel);
    });
    // check globalAudio
    addLog('info','globalAudio present? ' + (window.globalAudio ? 'YES' : 'NO'));
  }

    // Close
    panel.querySelector("#closeDebuggerBtn").addEventListener("click", togglePanel);

    // Tabs
    const tabs = panel.querySelectorAll("#customDebuggerTabs button");
    tabs.forEach(btn => {
      btn.addEventListener("click", () => {
        tabs.forEach(b => b.classList.remove("active"));
        btn.classList.add("active");

        const contentDivs = panel.querySelectorAll("#customDebuggerContent > div");
        contentDivs.forEach(div => (div.style.display = "none"));
        const target = panel.querySelector(#customDebugger${btn.dataset.tab.charAt(0).toUpperCase() + btn.dataset.tab.slice(1)});
        if (target) target.style.display = "block";
      });
    });

    // Example action: Render Week
    const renderBtn = panel.querySelector("#renderWeekBtn");
    if (renderBtn) {
      renderBtn.addEventListener("click", () => {
        console.log("[Debugger Action] Render Week triggered");
        // Hook into your real renderWeek() if available:
        if (typeof renderWeek === "function") renderWeek();
      });
    }

    return panel;
  }

  // ========== TOGGLE ==========
  function togglePanel() {
    if (!panel) createPanel();
    panelVisible = !panelVisible;
    panel.style.display = panelVisible ? "flex" : "none";
  }

  // ========== HOTKEY ==========
  document.addEventListener("keydown", (e) => {
    if (e.ctrlKey && e.key.toLowerCase() === "m") {
      e.preventDefault();
      togglePanel();
    }
  });

  // ========== SECRET TAP ==========
  let tapTimeout, tapCount = 0;
  document.body.addEventListener("click", (e) => {
    const nearBottomRight = (
      window.innerWidth - e.clientX < 60 &&
      window.innerHeight - e.clientY < 60
    );
    if (nearBottomRight) {
      tapCount++;
      clearTimeout(tapTimeout);
      tapTimeout = setTimeout(() => tapCount = 0, 400);
      if (tapCount >= 2) {
        togglePanel();
        tapCount = 0;
      }
    }
  });

  // ========== CONSOLE HOOK ==========
  const logContainer = () => panel?.querySelector("#customDebuggerLogs");
  ["log","warn","error"].forEach(level => {
    const orig = console[level];
    console[level] = (...args) => {
      orig.apply(console,args);
      const el = document.createElement("div");
      el.className = "debugger-log-entry";
      el.style.color = level === "error" ? "tomato" : level === "warn" ? "gold" : "white";
      el.textContent = [${level.toUpperCase()}] ${args.map(a => (typeof a==="object"?JSON.stringify(a):a)).join(" ")};
      logContainer()?.appendChild(el);
    };
  });

  console.log("[Debugger] Initialized. Press Ctrl+M or double-tap bottom-right to toggle.");

  // public API
  window.__dbg.addLog = addLog;
  // initial
  addLog('info','Sandbox ready — debug logging initialized');
  checkResources();
})();



</script>
