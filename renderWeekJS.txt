window.renderWeek = function (targetId) {
  const target = document.getElementById(targetId);
  if (!target) return;

  // Example "Week Data" (replace with real JSON later)
  const weekData = {
    torah: {
      title: "Torah Portion",
      ref: "Genesis 1:1–2:3",
      text: "In the beginning God created the heavens and the earth..."
    },
    prophets: {
      title: "Prophets",
      ref: "Isaiah 42:1–21",
      text: "Here is my servant, whom I uphold..."
    },
    writings: {
      title: "Writings",
      ref: "Psalm 23",
      text: "The Lord is my shepherd; I shall not want."
    },
    haftarah: {
      title: "Haftarah",
      ref: "Ezekiel 36:16–38",
      text: "I will give you a new heart and put a new spirit in you..."
    },
    gospel: {
      title: "Gospel",
      ref: "John 1:1–18",
      text: "In the beginning was the Word, and the Word was with God..."
    },
    notes: {
      title: "Study Notes",
      ref: "",
      text: "This week focuses on creation, covenant, and the Word made flesh."
    }
  };

  // Build sections dynamically
  target.innerHTML = ""; // clear old content
  Object.keys(weekData).forEach(key => {
    const { title, ref, text } = weekData[key];

    const card = document.createElement("div");
    card.className = "card";

    card.innerHTML = `
      <h3 class="text-xl font-bold mb-1">${title}</h3>
      <p class="text-sm text-gray-400 mb-2">${ref}</p>
      <p>${text}</p>
      <div class="mt-2 flex gap-2">
        <button class="px-2 py-1 bg-blue-500 text-white rounded hover:bg-blue-600">Read</button>
        <button class="px-2 py-1 bg-green-500 text-white rounded hover:bg-green-600">Listen</button>
      </div>
    `;

    target.appendChild(card);
  });
};
