<script>
(function(){
  const feastDays = {
    "2025-04-13": "Passover",
    "2025-06-01": "Pentecost",
    "2025-09-23": "Trumpets",
    "2025-10-02": "Day of Atonement",
    "2025-10-07": "Feast of Tabernacles"
  };

  const otherTraditions = {
    "Orthodox Easter": "2025-04-20",
    "Ramadan Start": "2025-03-30",
    "Diwali": "2025-11-01",
    "Chinese New Year": "2025-01-29"
  };

  const weekDisplay = document.getElementById("weekDisplay");
  const sabbathDisplay = document.getElementById("sabbathDisplay");
  const feastDisplayEl = document.getElementById("feastDisplay");
  const otherTraditionsEl = document.getElementById("otherTraditions");
  const holyDayCountdown = document.getElementById("holyDayCountdown");

  // Helpers
  function formatDate(d) { return d.toISOString().split("T")[0]; }

  function getWeek(date = new Date()) {
    const start = new Date(date); start.setDate(date.getDate() - date.getDay());
    return Array.from({length:7},(_,i)=>{
      const d = new Date(start); d.setDate(start.getDate()+i); return d;
    });
  }

  function getNextSabbath() {
    const today = new Date();
    const diff = (6 - today.getDay() + 7) % 7 || 7;
    const sabbath = new Date(today);
    sabbath.setDate(today.getDate() + diff);
    sabbath.setHours(18,0,0,0); // sunset approx 6pm
    return sabbath;
  }

  function getNextFeast() {
    const today = new Date();
    const futureFeasts = Object.keys(feastDays)
      .map(d=>new Date(d))
      .filter(d=>d>=today)
      .sort((a,b)=>a-b);
    return futureFeasts[0];
  }

  function updateCountdown(el, targetDate) {
    const now = new Date();
    const diff = targetDate - now;
    if(diff<=0){el.textContent="Now!"; return;}
    const days = Math.floor(diff/86400000);
    const hours = Math.floor((diff%86400000)/3600000);
    const minutes = Math.floor((diff%3600000)/60000);
    const seconds = Math.floor((diff%60000)/1000);
    el.textContent = ${days}d ${hours}h ${minutes}m ${seconds}s;
  }

  // Render Week
  function renderWeek() {
    const week = getWeek();
    weekDisplay.textContent = week.map(d=>formatDate(d)).join(" • ");
  }

  // Render Sabbath
  function renderSabbath() {
    const sab = getNextSabbath();
    updateCountdown(sabbathDisplay, sab);
  }

  // Render Feasts
  function renderFeasts() {
    feastDisplayEl.innerHTML = "";
    Object.entries(feastDays).forEach(([date,name])=>{
      const li = document.createElement("li");
      li.textContent = ${name}: ${date};
      feastDisplayEl.appendChild(li);
    });
    const nextFeast = getNextFeast();
    updateCountdown(holyDayCountdown, nextFeast);
  }

  // Other Traditions
  function renderOtherTraditions() {
    otherTraditionsEl.innerHTML = "";
    Object.entries(otherTraditions).forEach(([name,date])=>{
      const li = document.createElement("li");
      li.textContent = ${name}: ${date};
      otherTraditionsEl.appendChild(li);
    });
  }

  // Initial render
  renderWeek();
  renderSabbath();
  renderFeasts();
  renderOtherTraditions();

  // Update countdowns every second
  setInterval(()=>{
    renderSabbath();
    renderFeasts();
  },1000);

})();
</script>
